cmake_minimum_required(VERSION 3.25)
project(VoxelControls VERSION 0.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(VOXEL_BUILD_PLUGIN "Build the Windows SKSE plugin" ${WIN32})
option(VOXEL_PLAYTEST "Enable the local development command file (never for releases)" OFF)
add_library(voxel_core STATIC src/core.cpp)
target_include_directories(voxel_core PUBLIC include)
if(VOXEL_BUILD_PLUGIN)
  set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)
  add_subdirectory(deps/CommonLibSSE-NG)
  include(deps/CommonLibSSE-NG/cmake/CommonLibSSE.cmake)
  add_library(voxel_imgui STATIC
    deps/imgui/imgui.cpp deps/imgui/imgui_draw.cpp deps/imgui/imgui_tables.cpp
    deps/imgui/imgui_widgets.cpp deps/imgui/backends/imgui_impl_dx11.cpp
    deps/imgui/backends/imgui_impl_win32.cpp)
  target_include_directories(voxel_imgui PUBLIC deps/imgui deps/imgui/backends)
  target_link_libraries(voxel_imgui PUBLIC d3d11 dxgi d3dcompiler dwmapi)
  add_library(VoxelControls SHARED src/plugin.cpp src/overlay.cpp src/runtime.cpp src/player_model.cpp)
  target_precompile_headers(VoxelControls PRIVATE include/voxel/pch.hpp)
  target_link_libraries(VoxelControls PRIVATE CommonLibSSE::CommonLibSSE voxel_core voxel_imgui)
  target_include_directories(VoxelControls PRIVATE include)
  target_compile_definitions(VoxelControls PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(VOXEL_PLAYTEST)
    find_package(directxtk CONFIG REQUIRED)
    target_link_libraries(VoxelControls PRIVATE Microsoft::DirectXTK)
    target_compile_definitions(VoxelControls PRIVATE VOXEL_PLAYTEST)
  endif()
  add_executable(voxel_runtime_smoke tests/windows_runtime_smoke.cpp)
  find_package(spdlog CONFIG REQUIRED)
  target_link_libraries(voxel_runtime_smoke PRIVATE spdlog::spdlog)
  install(TARGETS VoxelControls RUNTIME DESTINATION SKSE/Plugins LIBRARY DESTINATION SKSE/Plugins)
else()
  enable_testing()
  add_executable(voxel_tests tests/core_tests.cpp)
  target_link_libraries(voxel_tests PRIVATE voxel_core)
  add_test(NAME voxel_core COMMAND voxel_tests)
  add_executable(voxel_control_tests tests/control_tests.cpp)
  target_include_directories(voxel_control_tests PRIVATE include)
  add_test(NAME voxel_controls COMMAND voxel_control_tests)
  add_executable(voxel_physics_reference_tests tests/physics_reference_tests.cpp)
  target_link_libraries(voxel_physics_reference_tests PRIVATE voxel_core)
  add_test(NAME voxel_minecraft_reference COMMAND voxel_physics_reference_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/minecraft-java-1.21.1.csv")
endif()
